graph TB
    subgraph "Frontend Layer"
        A[User enters Card Data
- Card Number
- CVV
- Expiry Date
- Cardholder Name] --> B[Payment Form]
        B --> C[POST Request to Backend
Full Card Data in Request Body]
    end

    subgraph "Backend Layer"
        C --> D[API Endpoint
Receives Card Data]
        D --> E{Validation}
        E -->|Invalid| F[Return Error Response
No Data Stored]
        E -->|Valid| G[Forward Card Data to
Payment Gateway]

        G --> H[Payment Gateway
Tokenization Service]
        H --> I{Tokenization
Response}

        I -->|Success| J[Receive Token Reference ID
+ Last 4 Digits]
        I -->|Failure| K[Return Error Response
No Data Stored]

        J --> L[Store in Database:
- Token Reference ID
- Last 4 Digits
- Metadata]
        L --> M[Return Success Response
to Frontend]
        K --> N[Return Failure Response
to Frontend]
    end

    subgraph "Database Layer"
        L --> O[(Payment Tokens Table)]
        O --> P[Columns:
- token_reference_id PK
- last_four_digits
- user_id FK
- created_at
- updated_at
- payment_gateway_id
- status]

        P -.->|Foreign Key| Q[(Users Table)]
        P -.->|Foreign Key| R[(Payment Gateways Table)]
    end

    subgraph "External Systems"
        H
    end

    subgraph "Data Retention Policy"
        S[Card Data in Memory:
- Exists only during request
- NOT persisted anywhere
- Cleared after response]
        T[Stored Data:
- Token Reference ID only
- Last 4 digits only
- No full card numbers
- No CVV
- No expiry date]
    end
